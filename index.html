-- Buat tabel jika belum ada CREATE TABLE IF NOT EXISTS products ( id BIGINT PRIMARY KEY, name TEXT NOT NULL, hpp NUMERIC DEFAULT 0, price NUMERIC NOT NULL, promo_price NUMERIC, discount NUMERIC DEFAULT 0, stock INT DEFAULT 0, img TEXT, points_earned INT DEFAULT 5, variants JSONB, tiers JSONB ); CREATE TABLE IF NOT EXISTS orders ( id TEXT PRIMARY KEY, display_date TEXT, name TEXT, phone TEXT, address TEXT, items JSONB, subtotal NUMERIC, total_hpp NUMERIC, total_discount NUMERIC, points_used INT, final_total NUMERIC, net_profit NUMERIC, points_earned INT, shipping TEXT, payment TEXT, status TEXT, paid_date TEXT ); CREATE TABLE IF NOT EXISTS customers ( phone TEXT PRIMARY KEY, name TEXT, address TEXT, points INT DEFAULT 0, total_spend NUMERIC DEFAULT 0 ); CREATE TABLE IF NOT EXISTS vouchers ( id BIGINT PRIMARY KEY, code TEXT, name TEXT, discount_amount NUMERIC, min_spend NUMERIC, max_usage INT, used_count INT DEFAULT 0 ); CREATE TABLE IF NOT EXISTS bundles ( id BIGINT PRIMARY KEY, name TEXT, p1 TEXT, p2 TEXT, p3 TEXT, p4 TEXT, p5 TEXT, original_price NUMERIC, promo_price NUMERIC, img TEXT, start_date TEXT, end_date TEXT ); CREATE TABLE IF NOT EXISTS tebus_murah ( id BIGINT PRIMARY KEY, name TEXT, normal_price NUMERIC, tebus_price NUMERIC, min_cart_amount NUMERIC, img TEXT, start_date TEXT, end_date TEXT ); CREATE TABLE IF NOT EXISTS store_settings ( id INT PRIMARY KEY, store_name TEXT, wa_admin TEXT, bank_accounts TEXT, admin_pin TEXT, point_value INT, logo_url TEXT );
